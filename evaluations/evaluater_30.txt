A 2x3 figure for the `iris` dataset:
  (1,1) box plot of sepal_length by species
  (1,2) manual violin of petal_length by species
  (1,3) joypy ridgeline of petal_width by species
  (2,1) correlation heatmap of numeric columns, divergent palette
        centered at 0
  (2,2) the same correlation matrix re-rendered with a 4-checkpoint
        quantile-anchored colormap
  (2,3) palette of three swatches representing the three species,
        derived from a triadic harmony starting at hue 0
Use plt.suptitle and tight_layout.


Assumptions:
- The `iris` dataset contains standard features: `sepal_length`, `sepal_width`, `petal_length`, `petal_width`, and `species`.
- The required libraries (`numpy`, `pandas`, `matplotlib.pyplot`, `seaborn`, `scipy`, `skimage`, `joypy`) are installed.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import gaussian_kde
from skimage import color, data as skimage_data
from matplotlib.colors import LinearSegmentedColormap, ListedColormap
import joypy

# Load the iris dataset
iris = sns.load_dataset('iris')
numeric_cols = iris.select_dtypes(include=np.number).columns

# --- Figure Setup ---
fig = plt.figure(figsize=(20, 10))
fig.suptitle("Data Visualization Lab 04: Iris Dataset Analysis", fontsize=16)
gs = fig.add_gridspec(2, 3)

# --- (1,1) Box plot of sepal_length by species ---
ax11 = fig.add_subplot(gs[0, 0])
sns.boxplot(data=iris, x='species', y='sepal_length', ax=ax11)
ax11.set_title('1.1 Sepal Length by Species (Box Plot)')
ax11.set_xlabel('Species')
ax11.set_ylabel('Sepal Length (cm)')

# --- (1,2) Manual violin of petal_length by species ---
ax12 = fig.add_subplot(gs[0, 1])
species_list = iris['species'].unique()
petal_lengths = [iris[iris['species'] == s]['petal_length'].values for s in species_list]

for i, vals in enumerate(petal_lengths):
    kde = gaussian_kde(vals)
    xs = np.linspace(vals.min(), vals.max(), 200)
    dens = kde(xs)
    # Normalize density and scale for visualization
    dens = dens / dens.max() * 0.4
    ax12.fill_betweenx(xs, i - dens, i + dens, alpha=0.4)

ax12.set_xticks(np.arange(len(species_list)))
ax12.set_xticklabels(species_list)
ax12.set_title('1.2 Petal Length by Species (Manual Violin)')
ax12.set_xlabel('Species')
ax12.set_ylabel('Petal Length (cm)')

# --- (1,3) joypy ridgeline of petal_width by species ---
ax13 = fig.add_subplot(gs[0, 2])
joypy.joyplot(iris, by='species', column='petal_width', kind='kde', overlap=0.5, ax=ax13)
ax13.set_title('1.3 Petal Width by Species (Ridgeline)')
ax13.set_xlabel('Petal Width (cm)')
ax13.set_ylabel('Species')

# --- (2,1) Correlation heatmap (Divergent Palette) ---
corr_matrix = iris[numeric_cols].corr()
ax21 = fig.add_subplot(gs[1, 0])

# Divergent palette: Blue (negative) -> White (zero) -> Red (positive)
divergent_cmap = LinearSegmentedColormap.from_list('bwrd', [(0,'blue'),(0.5,'white'),(1,'red')])

sns.heatmap(corr_matrix, cmap=divergent_cmap, center=0, ax=ax21)
ax21.set_title('2.1 Correlation Heatmap (Divergent)')
ax21.set_xlabel('Features')
ax21.set_ylabel('Features')

# --- (2,2) Correlation heatmap (Quantile-Anchored Colormap) ---
ax22 = fig.add_subplot(gs[1, 1])

# Quantile anchoring: 4 checkpoints (2, 50, 80)
M = corr_matrix.values
# Calculate quantiles for the matrix elements
q_low = np.quantile(M, 0.02)
q_mid = np.quantile(M, 0.5)
q_high = np.quantile(M, 0.98)

# Define stops for the colormap: (normalized_value, color)
# We use the quantiles to define the boundaries of the color map
# Normalized values: (q - vmin) / (vmax - vmin)
vmin = M.min()
vmax = M.max()

# 4 checkpoints: 2nd percentile, 50th percentile, 98th percentile, and min/max boundaries
stops = [
    (0.0, 'blue'),  # Min boundary
    ((q_low - vmin) / (vmax - vmin), 'blue'), # 2nd percentile
    ((q_mid - vmin) / (vmax - vmin), 'white'), # Median
    ((q_high - vmin) / (vmax - vmin), 'red'), # 98th percentile
    (1.0, 'red')    # Max boundary
]

custom_cmap = LinearSegmentedColormap.from_list('quantile_map', stops)

sns.heatmap(corr_matrix, cmap=custom_cmap, ax=ax22)
ax22.set_title('2.2 Correlation Heatmap (Quantile-Anchored)')
ax22.set_xlabel('Features')
ax22.set_ylabel('Features')

# --- (2,3) Triadic Color Palette ---
ax23 = fig.add_subplot(gs[1, 2])
species_names = iris['species'].unique()
species_hues = {s: i for i, s in enumerate(species_names)}

# Triadic harmony starting at hue 0: [0, 1/3, 2/3]
triadic_hues = [0.0, 1/3, 2/3]

# Build palette: convert HSV to RGB
cols = [color.hsv2rgb([hh, 0.65, 0.9]) for hh in triadic_hues]

# Map colors to species (assuming species order matches triadic order for simplicity)
# If species order is different, we map the colors to the unique species names
palette_colors = {
    species_names[0]: cols[0],
    species_names[1]: cols[1],
    species_names[2]: cols[2]
}

# Create swatches
swatch_row = []
for i, color_rgb in enumerate(cols):
    # Create a rectangle patch
    patch = plt.Rectangle((i, 0), 1, 1, color=color_rgb)
    swatch_row.append(patch)

# Add patches to the subplot
ax23.add_patch(swatch_row[0])
ax23.add_patch(swatch_row[1])
ax23.add_patch(swatch_row[2])

ax23.set_xlim(0, 3)
ax23.set_ylim(0, 1)
ax23.axis('off')
ax23.set_title('2.3 Triadic Color Palette (Hue 0, 1/3, 2/3)')

plt.tight_layout()
plt.show()
```

Notes:
- The manual violin plot uses `scipy.stats.gaussian_kde` to estimate the probability density function for each species' petal length.
- The quantile-anchored heatmap uses `np.quantile` to define color stops, ensuring the color mapping is anchored to specific data percentiles rather than just the global min/max.
- Triadic harmony was achieved by setting the hue values at 0, 1/3, and 2/3 in the [0, 1] range, which is necessary for `skimage` color functions.

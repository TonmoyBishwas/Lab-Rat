3)Assumptions:
- The base hue $h=0.6$ is provided.
- Saturation (S) and Value (V) are set to standard values (0.65, 0.9) for color generation.

```python
import numpy as np
import matplotlib.pyplot as plt
import skimage.color

# Define base hue and parameters
h_base = 0.6
S = 0.65
V = 0.9

# 1. Calculate complementary hue (h + 0.5)
h_complementary = np.mod(h_base + 0.5, 1.0)

# 2. Convert hues to RGB using HSV -> RGB
# For the base color
hsv_base = np.array([h_base, S, V])
rgb_base = skimage.color.hsv2rgb(hsv_base)

# For the complementary color
hsv_comp = np.array([h_complementary, S, V])
rgb_comp = skimage.color.hsv2rgb(hsv_comp)

# 3. Display colors as side-by-side rectangles
fig, ax = plt.subplots(1, 1, figsize=(5, 2))

# Draw the base color rectangle
rect_base = plt.Rectangle((0, 0), 1, 1, color=rgb_base)
ax.add_patch(rect_base)

# Draw the complementary color rectangle
rect_comp = plt.Rectangle((1, 0), 1, 1, color=rgb_comp)
ax.add_patch(rect_comp)

# Set plot properties
ax.set_xlim(0, 2)
ax.set_ylim(0, 1)
ax.axis('off')
ax.set_title("Complementary")

plt.tight_layout()
plt.show()
```

Notes:
- The complementary hue is calculated by adding 0.5 to the base hue and wrapping the result using `np.mod(h, 1.0)`.
- `plt.Rectangle` is used to draw the colors, where the first rectangle starts at x=0 and the second starts at x=1.
- Ensure S and V are kept constant when generating the palette to maintain consistent brightness and saturation.

8)Assumptions:
- The `iris` dataset is available and contains features including `petal_length` and a categorical species column.
- The species are 'setosa', 'versicolor', and 'virginica'.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.stats import gaussian_kde
from sklearn.datasets import load_iris

# Load the iris dataset
iris = load_iris()
df = pd.DataFrame(data=iris.data, columns=iris.feature_names)
df['species'] = iris.target_names[iris.target]

# Define the groups and the value column
groups = df['species'].unique()
value_col = 'petal length (cm)'

# Set up the figure and axes
fig, ax = plt.subplots(figsize=(10, 6))

# Determine the range of petal lengths for consistent plotting
min_val = df[value_col].min()
max_val = df[value_col].max()
x_range = np.linspace(min_val, max_val, 500)

# Plotting the manual violin plots
for i, group in enumerate(groups):
    # 1. Extract data for the current group
    group_data = df[df['species'] == group][value_col]

    # 2. Calculate KDE
    kde = gaussian_kde(group_data)

    # 3. Calculate density values over the defined range
    dens = kde(x_range)

    # 4. Plot the mirrored density using fill_betweenx
    # The density is plotted around the position 'i'
    ax.fill_betweenx(i, dens, alpha=0.4, label=group)

# Styling the plot
ax.set_xticks(range(len(groups)))
ax.set_xticklabels(groups)
ax.set_xlabel("Species")
ax.set_ylabel("Density of Petal Length")
ax.set_title("Manual Violin Plot of Petal Length by Iris Species")
ax.grid(axis='y', linestyle='--', alpha=0.7)

plt.tight_layout()
plt.show()
```

Notes:
- The density is calculated using `gaussian_kde` and then plotted using `fill_betweenx` centered at the index `i` corresponding to the species group.
- The x-axis ticks are set to represent the species names for clear labeling.
- If you were to use a different feature (e.g., sepal width), simply change `value_col` accordingly.

```python
import numpy as np
import matplotlib.pyplot as plt
import skimage.color
from matplotlib.patches import Rectangle

# --- Setup ---
base_hue = 0.08
S = 0.65  # Saturation
V = 0.9   # Value

# --- Harmony Calculations ---

# 1. Complementary (2 hues)
h_comp = np.array([base_hue, np.mod(base_hue + 0.5, 1.0)])
colors_comp = [skimage.color.hsv2rgb([h, S, V]) for h in h_comp]

# 2. Triadic (3 hues)
h_tri = np.array([base_hue, np.mod(base_hue + 1/3, 1.0), np.mod(base_hue + 2/3, 1.0)])
colors_tri = [skimage.color.hsv2rgb([h, S, V]) for h in h_tri]

# 3. Analogous (3 hues)
h_ana = np.array([base_hue, np.mod(base_hue + 1/12, 1.0), np.mod(base_hue - 1/12, 1.0)])
colors_ana = [skimage.color.hsv2rgb([h, S, V]) for h in h_ana]

# --- Plotting ---
fig, axes = plt.subplots(3, 1, figsize=(8, 6), sharex=True)
fig.suptitle("Color Harmony Palettes based on Base Hue h=0.08", fontsize=14)

# Helper function to draw swatches
def draw_swatches(ax, colors, title):
    ax.set_title(title, fontsize=12)
    swatch_rects = []
    for i, color in enumerate(colors):
        # Create a rectangle patch
        rect = Rectangle((i, 0), 1, 1, color=color)
        ax.add_patch(rect)
        swatch_rects.append(rect)
    
    # Set limits and remove axis clutter
    ax.set_xlim(-0.5, len(colors) - 0.5)
    ax.set_ylim(0, 1)
    ax.axis('off')
    return ax

# Plot Complementary
ax0 = draw_swatches(axes[0], colors_comp, "Complementary Harmony")

# Plot Triadic
ax1 = draw_swatches(axes[1], colors_tri, "Triadic Harmony")

# Plot Analogous
ax2 = draw_swatches(axes[2], colors_ana, "Analogous Harmony")

plt.tight_layout(rect=[0, 0.03, 1, 0.95]) # Adjust layout for suptitle
plt.show()
```

Notes:
- The `Rectangle` patch is used to represent the color swatches within each subplot.
- `plt.tight_layout` is crucial here to prevent the subplot titles from overlapping the main figure title.
- Ensure S and V are kept constant across all harmonies to compare the hue relationships accurately.

```python
])

plt.figure(figsize=(8, 6))

    corr_matrix,
    annot=True,           # Annotate cells with correlation values
    cmap=custom_cmap,     # Apply the custom divergent colormap
    center=0,             # Center the colormap at 0
    fmt=".2f",            # Format annotations to two decimal places
    linewidths=.5,        # Add lines between cells for clarity
    cbar_kws={'label': 'Correlation Coefficient'}
```

```python
# 1. Generate the 30x30 random matrix

outlier_rows = rng.integers(0, 30, size=num_outliers, dtype=int)
outlier_cols = rng.integers(0, 30, size=num_outliers, dtype=int)

for i in range(num_outliers):
    r, c = outlier_rows[i], outlier_cols[i]
    # Inject a value around 12 (e.g., 12 + small random noise)
    M_raw[r, c] = 12 + rng.normal(0, 0.5)

# 3. Calculate percentiles and clip the matrix
lower_bound = np.quantile(M_raw, 0.02)
upper_bound = np.quantile(M_raw, 0.98)
M_clipped = np.clip(M_raw, lower_bound, upper_bound)

# 4. Plotting
fig, axes = plt.subplots(1, 2, figsize=(14, 6))
plt.suptitle("Heatmap Comparison: Raw vs. Clipped Matrix", fontsize=16)

# Raw Heatmap (Left)
sns.heatmap(M_raw, ax=axes[0], cmap="viridis", cbar_kws={'label': 'Value'})
axes[0].set_title("Raw Matrix (with Outliers)")
axes[0].set_xlabel("Column Index")
axes[0].set_ylabel("Row Index")
```

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import gaussian_kde
try:
    import joypy
except ImportError:
    print("joypy not found. Using a simplified ridgeline approach.")
    joypy = None

# --- Data Synthesis ---
N_per_species = 80
np.random.seed(42)

# Species 1: Normal distribution
data_s1 = np.random.normal(loc=10, scale=2, size=N_per_species)

# Species 2: Slightly skewed distribution
data_s2 = np.random.normal(loc=15, scale=3, size=N_per_species)

# Species 3: Bimodal distribution (two distinct peaks)
data_s3_peak1 = np.random.normal(loc=8, scale=1.5, size=N_per_species // 2)
data_s3_peak2 = np.random.normal(loc=18, scale=2, size=N_per_species // 2)
data_s3 = np.concatenate([data_s3_peak1, data_s3_peak2])

# Species 4: Uniform distribution
data_s4 = np.random.uniform(low=5, high=25, size=N_per_species)

# Create DataFrame
df = pd.DataFrame({
    'Weight': np.concatenate([data_s1, data_s2, data_s3, data_s4]),
    'Species': ['S1'] * N_per_species + ['S2'] * N_per_species + ['S3'] * N_per_species + ['S4'] * N_per_species
})

# --- Plotting ---
fig, axes = plt.subplots(1, 3, figsize=(20, 6))
fig.suptitle("Weight Distributions Across Four Fish Species", fontsize=16)

# 1. Box Plot
sns.boxplot(data=df, x='Species', y='Weight', ax=axes[0])
axes[0].set_title('Box Plot')
axes[0].set_xlabel('Species')
axes[0].set_ylabel('Weight')

# 2. Manual Violin Plot (using KDE)
# We iterate over the species groups to plot the KDE manually
for i, species in enumerate(df['Species'].unique()):
    vals = df[df['Species'] == species]['Weight'].values
    
    # Calculate KDE
    kde = gaussian_kde(vals)
    xs = np.linspace(vals.min(), vals.max(), 200)
    dens = kde(xs)
    
    # Normalize density and scale for visualization
    dens = dens / dens.max() * 0.4
    
    # Plot the density curve (violin shape)
    axes[1].fill_betweenx(xs, i - dens, i + dens, alpha=0.5, label=species)

axes[1].set_title('Manual Violin Plot (KDE)')
axes[1].set_xticks(range(len(df['Species'].unique())))
axes[1].set_xticklabels(df['Species'].unique())
axes[1].set_xlabel('Species')
axes[1].set_ylabel('Weight')
axes[1].legend(loc='upper right')


# 3. Ridgeline Plot
if joypy:
    # Use joypy for the ridgeline plot
    joypy.joyplot(df, by='Species', column='Weight', kind='kde', overlap=0.5, ax=axes[2])
    axes[2].set_title('Ridgeline Plot (Joypy)')
```

29)Assumptions:
- The reaction times are positive values, suitable for log transformation.
- The synthetic `condition` column has exactly 3 levels.
- The total dataset size is 500 rows.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import gaussian_kde

# --- Data Synthesis ---
N_total = 500
N_conditions = 3
N_per_condition = N_total // N_conditions

# 1. Generate base skewed data (e.g., using an exponential distribution)
np.random.seed(42)
base_rt = np.random.exponential(scale=0.1, size=N_total)

# 2. Create the condition column
conditions = np.repeat([1, 2, 3], N_per_condition)
df = pd.DataFrame({'condition': conditions, 'reaction_time': base_rt})

# 3. Inject outliers (10% of total data, 50 points)
n_outliers = int(0.10 * N_total)
outlier_indices = np.random.choice(df.index, n_outliers, replace=False)
# Outliers are 3x to 5x larger than the median of the base data
median_rt = df['reaction_time'].median()
outlier_values = median_rt * np.random.uniform(3, 5, n_outliers)
df.loc[outlier_indices, 'reaction_time'] = outlier_values

# --- Data Transformation ---
# Transformation: sign(x) * log1p(|x|)
# Since reaction_time > 0, this simplifies to np.log1p(reaction_time)
df['transformed_rt'] = np.log1p(df['reaction_time'])

# --- Plotting Setup ---
fig, axes = plt.subplots(2, 3, figsize=(18, 10))
plt.suptitle("Distribution Analysis: Raw vs. Log-Transformed Reaction Times", fontsize=16)

# Function to plot the three distribution types for a given dataframe
def plot_distributions(df, row_index, title_suffix):
    
    # 1. Box Plot (Seaborn)
    sns.boxplot(data=df, x='condition', y='reaction_time' if row_index == 0 else 'transformed_rt', ax=axes[row_index, 0])
    axes[row_index, 0].set_title(f'Box Plot ({title_suffix})')
    axes[row_index, 0].set_xlabel('Condition')
    axes[row_index, 0].set_ylabel('Reaction Time')

    # 2. Manual Violin Plot (KDE density)
    for i in range(N_conditions):
        group_data = df[df['condition'] == i + 1]['reaction_time' if row_index == 0 else 'transformed_rt']
        if len(group_data) > 0:
            kde = gaussian_kde(group_data)
            xs = np.linspace(group_data.min(), group_data.max(), 200)
            dens = kde(xs)
            # Normalize density and scale for visualization
            dens = dens / dens.max() * 0.4 
            
            # Plot density centered at position i+1
            axes[row_index, 1].fill_between(xs, i - dens, i + dens, alpha=0.4, label=f'C{i+1}')
            axes[row_index, 1].set_xticks(np.arange(1, N_conditions + 1))
            axes[row_index, 1].set_xticklabels([f'C{i+1}' for i in range(N_conditions)])
            axes[row_index, 1].set_title(f'Manual Violin Plot ({title_suffix})')
            axes[row_index, 1].set_ylabel('Density')

    # 3. Ridgeline Plot (Manual Stacking of KDEs)
    # We use the same KDEs calculated above, but stack them vertically
    y_offset = 0
    for i in range(N_conditions):
        group_data = df[df['condition'] == i + 1]['reaction_time' if row_index == 0 else 'transformed_rt']
        if len(group_data) > 0:
            kde = gaussian_kde(group_data)
            xs = np.linspace(group_data.min(), group_data.max(), 200)
            dens = kde(xs)
            dens = dens / dens.max() * 0.4
            
            # Plot the density curve shifted by y_offset
            axes[row_index, 2].fill_between(xs, y_offset - dens, y_offset + dens, alpha=0.4)
            y_offset += 0.8 # Increase offset for next curve
            
    axes[row_index, 2].set_title(f'Ridgeline Plot ({title_suffix})')
    axes[row_index, 2].set_xlabel('Reaction Time')
    axes[row_index, 2].set_ylabel('Condition')
    
    # General cleanup for the row
    for ax in axes[row_index, :]:
        ax.grid(axis='y', linestyle='--', alpha=0.6)


# Row 1: Raw Data
plot_distributions(df, 0, "Raw Data")

# Row 2: Transformed Data
plot_distributions(df, 1, "Transformed Data (log1p)")

plt.tight_layout()
plt.show()
```

Notes:
- The **Box Plot** is the best plot to reveal the right skew and the presence of extreme outliers in the raw data due to its explicit display of quartiles and whiskers.
- The **Ridgeline Plot** is the most robust to the transformation because it visualizes the underlying probability density functions (KDEs) rather than relying on summary statistics (like the median/IQR in a box plot).
- The transformation successfully compresses the spread of the raw data, making the distributions in the bottom row appear much more symmetrical and less influenced by the extreme outliers.

```python
iris = sns.load_dataset('iris')
```

Assumptions:
- The `iris` dataset contains columns: `sepal_length`, `sepal_width`, `petal_length`, `petal_width`, and `species`.
- `joypy` is available for the ridgeline plot.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import gaussian_kde
from skimage import color, img_as_float
import joypy # Required for ridgeline plot
from matplotlib.colors import ListedColormap

# Load the iris dataset
iris = sns.load_dataset('iris')

# Setup figure and axes (2 rows, 3 columns)
fig, axes = plt.subplots(2, 3, figsize=(20, 10))
fig.suptitle("Data Visualization Lab 04: Spread Plots, Ridgeline, and Color Theory", fontsize=16)

# --- (1,1) Box plot of sepal_length by species ---
sns.boxplot(data=iris, x='species', y='sepal_length', ax=axes[0, 0])
axes[0, 0].set_title('1.1 Sepal Length by Species (Box Plot)')
axes[0, 0].set_xlabel('Species')
axes[0, 0].set_ylabel('Sepal Length')

# --- (1,2) Manual violin of petal_length by species ---
species_list = iris['species'].unique()
for i, species in enumerate(species_list):
    vals = iris[iris['species'] == species]['petal_length']
    
    # Calculate KDE
    kde = gaussian_kde(vals)
    xs = np.linspace(vals.min(), vals.max(), 200)
    dens = kde(xs)
    
    # Normalize and scale for visualization
    dens = dens / dens.max() * 0.4
    
    # Plot the violin shape
    plt.fill_betweenx(xs, i - dens, i + dens, alpha=0.4, ax=axes[0, 1])

axes[0, 1].set_title('1.2 Petal Length by Species (Manual Violin)')
axes[0, 1].set_xticks(range(len(species_list)))
axes[0, 1].set_xticklabels(species_list)
axes[0, 1].set_xlabel('Species')
axes[0, 1].set_ylabel('Petal Length')


# --- (1,3) Ridgeline of petal_width by species ---
joypy.joyplot(iris, by='species', column='petal_width', ax=axes[0, 2])
axes[0, 2].set_title('1.3 Petal Width by Species (Ridgeline)')
axes[0, 2].set_xlabel('Species')
axes[0, 2].set_ylabel('Petal Width')


# --- (2,1) Correlation heatmap (Divergent Palette) ---
numeric_cols = iris.select_dtypes(include=np.number).columns
corr_matrix = iris[numeric_cols].corr()

sns.heatmap(corr_matrix, annot=True, cmap='bwrd', center=0, linewidths=.5, linecolor='black', ax=axes[1, 0])
axes[1, 0].set_title('2.1 Correlation Heatmap (Divergent Palette)')
axes[1, 0].set_xlabel('Features')
axes[1, 0].set_ylabel('Features')


# --- (2,2) Correlation heatmap (Quantile-anchored Colormap) ---
# Calculate ranks for quantile mapping
M = corr_matrix.values
ranks = M.flatten().argsort().argsort() / (M.size - 1)
M_q = ranks.reshape(M.shape)

# Use the ranked matrix for the heatmap
sns.heatmap(M_q, annot=True, cmap='viridis', linewidths=.5, linecolor='black', ax=axes[1, 1])
axes[1, 1].set_title('2.2 Correlation Heatmap (Quantile-Anchored)')
axes[1, 1].set_xlabel('Features (Quantile Rank)')
axes[1, 1].set_ylabel('Features (Quantile Rank)')


# --- (2,3) Color Palette (Triadic Harmony) ---
# Triadic harmony starting at hue 0: [0, 1/3, 2/3]
hues = [0.0, 1/3, 2/3]
# Use a fixed saturation and value for consistency
S = 0.65
V = 0.9
cols = [color.hsv2rgb([h, S, V]) for h in hues]

ax = axes[1, 2]
ax.add_patch(plt.Rectangle((0, 0), 1, 1, color=cols[0]))
ax.add_patch(plt.Rectangle((1, 0), 1, 1, color=cols[1]))
ax.add_patch(plt.Rectangle((2, 0), 1, 1, color=cols[2]))

ax.set_xlim(0, 3)
ax.set_ylim(0, 1)
ax.axis('off')
ax.set_title('2.3 Triadic Color Palette (h=0, 1/3, 2/3)')

# Final layout adjustments
plt.tight_layout()
plt.show()
```

Notes:
- For the manual violin plot, the KDE is calculated for each species group and scaled to fit the subplot area.
- The quantile heatmap uses the rank of the correlation values across the entire matrix to create a perceptually uniform color scale.
- The triadic palette uses `skimage.color.hsv2rgb` with fractional hues (0, 1/3, 2/3) to generate the colors.
